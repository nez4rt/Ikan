# Konfigurasi Abbr Fish Shell berbagai distro

```fish
# ==========================================
# UBUNTU (Menggunakan Nala)
# ==========================================
abbr -a ff "fastfetch"
abbr -a cls "clear"
abbr -a nari "nala search"
abbr -a nasa "sudo nala install"

# ==========================================
# FEDORA (Menggunakan DNF)
# ==========================================
abbr -a ff "fastfetch"
abbr -a cls "clear"
abbr -a nari "dnf se"
abbr -a nasa "sudo dnf install"

# ==========================================
# ARCH LINUX (Menggunakan Pacman & Paru)
# ==========================================
abbr -a ff "fastfetch"
abbr -a cls "clear"
abbr -a pari "pacman -Ss"
abbr -a pasang "sudo pacman -S"
abbr -a yari "paru -Ss"
abbr -a yasa "paru -S"
